# rate5: 6000 numbered lines at once (100 bytes each), then 5 lines/s forever.
apiVersion: v1
kind: Pod
metadata: {name: rate5, namespace: proof, labels: {proof: "true"}}
spec:
  restartPolicy: Never
  containers:
  - name: rate5
    image: busybox:1.37.0
    imagePullPolicy: IfNotPresent
    command: ["sh", "-c"]
    args:
    - |
      P=$(head -c 87 /dev/zero | tr '\0' 'x')
      i=1
      while [ $i -le 6000 ]; do printf 'rate5 %06d %s\n' $i "$P"; i=$((i+1)); done
      while true; do printf 'rate5 %06d %s\n' $i "$P"; i=$((i+1)); sleep 0.2; done
---

# web: a Deployment of 2 replicas x 2 containers. app: 3000 lines at once, then 5 lines/s; side: 200 lines, then silent.
apiVersion: apps/v1
kind: Deployment
metadata: {name: web, namespace: proof}
spec:
  replicas: 2
  selector: {matchLabels: {app: web}}
  template:
    metadata: {labels: {app: web}}
    spec:
      containers:
      - name: app
        image: busybox:1.37.0
        imagePullPolicy: IfNotPresent
        command: ["sh", "-c"]
        args:
        - |
          P=$(head -c 87 /dev/zero | tr '\0' 'w')
          i=1
          while [ $i -le 3000 ]; do printf 'web-app %06d %s\n' $i "$P"; i=$((i+1)); done
          while true; do printf 'web-app %06d %s\n' $i "$P"; i=$((i+1)); sleep 0.2; done
      - name: side
        image: busybox:1.37.0
        imagePullPolicy: IfNotPresent
        command: ["sh", "-c"]
        args:
        - |
          i=1
          while [ $i -le 200 ]; do printf 'web-side %06d side line\n' $i; i=$((i+1)); done
          sleep 1000000
