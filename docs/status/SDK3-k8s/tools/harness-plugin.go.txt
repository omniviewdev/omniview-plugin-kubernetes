// Package harness launches the Kubernetes plugin binary the way the Omniview host does (go-plugin,
// the SDK handshake, AutoMTLS, the broker multiplexed on the process listener, the SDK's dial
// options) and talks to it with the SDK's own client types: the lifecycle, resource v2 and log
// capabilities. It owns a throwaway HOME so that nothing of the user's ~/.omniview or ~/.kube is
// read or written.
//
// This is test tooling (zz-proof); it is not part of the plugin.
package harness

import (
	"bytes"
	"context"
	"fmt"
	"io"
	"os"
	"os/exec"
	"path/filepath"
	"strings"
	"sync"
	"time"

	"github.com/google/uuid"
	"github.com/hashicorp/go-hclog"
	goplugin "github.com/hashicorp/go-plugin"
	"google.golang.org/grpc"

	"github.com/omniviewdev/plugin-sdk/pkg/config"
	resourcev2 "github.com/omniviewdev/plugin-sdk/pkg/resource/v2"
	resourceplugin "github.com/omniviewdev/plugin-sdk/pkg/resource/v2/plugin"
	"github.com/omniviewdev/plugin-sdk/pkg/sdk"
	"github.com/omniviewdev/plugin-sdk/pkg/types"
	ep "github.com/omniviewdev/plugin-sdk/pkg/v1/exec"
	lc "github.com/omniviewdev/plugin-sdk/pkg/v1/lifecycle"
	lp "github.com/omniviewdev/plugin-sdk/pkg/v1/logs"
	mp "github.com/omniviewdev/plugin-sdk/pkg/v1/metric"
	np "github.com/omniviewdev/plugin-sdk/pkg/v1/networker"
	sp "github.com/omniviewdev/plugin-sdk/pkg/v1/settings"
	tp "github.com/omniviewdev/plugin-sdk/pkg/v1/tools"
)

// PluginID is the id the plugin is installed under; it is also the magic cookie value.
const PluginID = "kubernetes"

// LaunchOptions describes one plugin process.
type LaunchOptions struct {
	// Name labels the process in files and tables ("base", "bump", "head").
	Name string
	// Binary is the plugin executable.
	Binary string
	// PluginYAML is the plugin.yaml of the plugin tree; the SDK reads it from the plugin's state directory.
	PluginYAML string
	// Kubeconfig is copied to $HOME/.kube/config: the plugin's default `kubeconfigs` setting.
	Kubeconfig string
	// Dir is the scratch directory of this process (HOME, state, logs). It is created, and must not exist yet.
	Dir string
	// LogLevel is the plugin's initial log level (OMNIVIEW_PLUGIN_LOG_LEVEL: debug, info, ...); empty keeps the default.
	LogLevel string
	// StartTimeout bounds the go-plugin handshake (the host uses 15 s).
	StartTimeout time.Duration
	// ExtraEnv is appended to the child's environment.
	ExtraEnv []string
}

// Plugin is a running plugin process and its dispensed capabilities.
type Plugin struct {
	Name       string
	Dir        string
	Home       string
	Binary     string
	Launched   time.Time // when Launch began
	Ready      time.Time // when the go-plugin handshake and the dispense were done
	StderrPath string    // the plugin's stderr, every line stamped with the harness clock
	HostLog    string    // go-plugin's own log (hclog, debug)

	Client    *goplugin.Client
	RPC       goplugin.ClientProtocol
	Conn      *grpc.ClientConn
	Lifecycle *lc.Client
	Resource  *resourceplugin.Client
	Logs      lp.Provider

	stderr *stampWriter
	hostFD *os.File
	once   sync.Once
}

// Launch starts the plugin as the host does (backend/pkg/plugin/loader.go createBackend).
func Launch(opts LaunchOptions) (*Plugin, error) {
	if opts.StartTimeout == 0 {
		opts.StartTimeout = 15 * time.Second
	}
	abs := func(p string) (string, error) { return filepath.Abs(p) }
	dir, err := abs(opts.Dir)
	if err != nil {
		return nil, err
	}
	if _, err := os.Stat(dir); err == nil {
		return nil, fmt.Errorf("scratch dir %s exists", dir)
	}
	home := filepath.Join(dir, "home")
	state := filepath.Join(home, ".omniview")
	pluginDir := filepath.Join(state, "plugins", PluginID)
	for _, d := range []string{filepath.Join(home, ".kube"), pluginDir} {
		if err := os.MkdirAll(d, 0o700); err != nil {
			return nil, err
		}
	}
	if err := copyFile(opts.Kubeconfig, filepath.Join(home, ".kube", "config"), 0o600); err != nil {
		return nil, fmt.Errorf("kubeconfig: %w", err)
	}
	if err := copyFile(opts.PluginYAML, filepath.Join(pluginDir, "plugin.yaml"), 0o600); err != nil {
		return nil, fmt.Errorf("plugin.yaml: %w", err)
	}
	binary, err := abs(opts.Binary)
	if err != nil {
		return nil, err
	}

	// The host's child environment (internal/instance Profile.ChildEnvironment): the host's own
	// environment plus GOWORK=off and the OMNIVIEW_* identity of the instance. Here HOME is the
	// scratch one, KUBECONFIG is dropped (the plugin must read only $HOME/.kube/config) and the
	// OTLP variables are dropped as the host does.
	var env []string
	for _, e := range os.Environ() {
		k, _, _ := strings.Cut(e, "=")
		switch {
		case k == "HOME", k == "KUBECONFIG", k == "GOWORK", k == "OMNIVIEW_PLUGIN_LOG_LEVEL",
			strings.HasPrefix(k, "OMNIVIEW_"), strings.HasPrefix(k, "OTEL_EXPORTER_OTLP_"):
			continue
		}
		env = append(env, e)
	}
	env = append(env,
		"HOME="+home,
		"GOWORK=off",
		"OMNIVIEW_STATE_DIR="+state,
		"OMNIVIEW_INSTANCE_ID="+uuid.NewString(),
		"OMNIVIEW_GENERATION="+uuid.NewString(),
		"OMNIVIEW_WORKSPACE_ID=",
		"OMNIVIEW_PLUGIN_ID="+PluginID,
	)
	if opts.LogLevel != "" {
		env = append(env, sdk.LogLevelEnvironment+"="+opts.LogLevel)
	}
	env = append(env, opts.ExtraEnv...)

	p := &Plugin{Name: opts.Name, Dir: dir, Home: home, Binary: binary, Launched: time.Now(),
		StderrPath: filepath.Join(dir, "plugin-stderr.log"), HostLog: filepath.Join(dir, "go-plugin-hclog.log")}
	errFD, err := os.Create(p.StderrPath)
	if err != nil {
		return nil, err
	}
	p.stderr = &stampWriter{w: errFD, closer: errFD}
	p.hostFD, err = os.Create(p.HostLog)
	if err != nil {
		return nil, err
	}

	meta := config.PluginMeta{ID: PluginID}
	cmd := exec.CommandContext(context.Background(), binary)
	cmd.Env = env
	p.Client = goplugin.NewClient(&goplugin.ClientConfig{
		HandshakeConfig: meta.GenerateHandshakeConfig(),
		VersionedPlugins: map[int]goplugin.PluginSet{
			1: {
				"resource":  &resourceplugin.GRPCPlugin{},
				"exec":      &ep.Plugin{Plugin: nil, Impl: nil},
				"networker": &np.Plugin{Plugin: nil, Impl: nil},
				"log":       &lp.Plugin{Plugin: nil, Impl: nil},
				"metric":    &mp.Plugin{Plugin: nil, Impl: nil},
				"settings":  &sp.SettingsPlugin{Plugin: nil, Impl: nil},
				"tools":     &tp.Plugin{Plugin: nil, Impl: nil},
				"lifecycle": &lc.Plugin{Plugin: nil, Impl: nil},
			},
		},
		GRPCBrokerMultiplex: true,
		AutoMTLS:            true,
		GRPCDialOptions:     sdk.GRPCDialOptions(),
		Cmd:                 cmd,
		SkipHostEnv:         true,
		StartTimeout:        opts.StartTimeout,
		AllowedProtocols:    []goplugin.Protocol{goplugin.ProtocolGRPC},
		Stderr:              p.stderr,
		Logger: hclog.New(&hclog.LoggerOptions{
			Name:   PluginID,
			Output: p.hostFD,
			Level:  hclog.Debug,
		}),
	})
	rpc, err := p.Client.Client()
	if err != nil {
		p.Client.Kill()
		p.closeFiles()
		return nil, fmt.Errorf("launch %s: %w", opts.Name, err)
	}
	p.RPC = rpc
	gc, ok := rpc.(*goplugin.GRPCClient)
	if !ok {
		p.Client.Kill()
		p.closeFiles()
		return nil, fmt.Errorf("launch %s: client is %T, not a gRPC client", opts.Name, rpc)
	}
	p.Conn = gc.Conn
	for name, dst := range map[string]func(any) bool{
		"lifecycle": func(v any) bool { c, ok := v.(*lc.Client); p.Lifecycle = c; return ok },
		"resource":  func(v any) bool { c, ok := v.(*resourceplugin.Client); p.Resource = c; return ok },
		"log":       func(v any) bool { c, ok := v.(lp.Provider); p.Logs = c; return ok },
	} {
		raw, derr := rpc.Dispense(name)
		if derr != nil {
			p.Close()
			return nil, fmt.Errorf("dispense %s: %w", name, derr)
		}
		if !dst(raw) {
			p.Close()
			return nil, fmt.Errorf("dispense %s: unexpected type %T", name, raw)
		}
	}
	p.Ready = time.Now()
	return p, nil
}

// Close kills the plugin process (the host's Kill) and closes the files. Safe to call twice.
func (p *Plugin) Close() {
	p.once.Do(func() {
		if p.Client != nil {
			p.Client.Kill()
		}
		p.closeFiles()
	})
}

func (p *Plugin) closeFiles() {
	if p.stderr != nil {
		p.stderr.Close()
	}
	if p.hostFD != nil {
		_ = p.hostFD.Close()
	}
}

// StderrLines returns the plugin's stderr as lines, each prefixed by the harness clock ("2026-..Z | <line>").
func (p *Plugin) StderrLines() []string {
	data, err := os.ReadFile(p.StderrPath)
	if err != nil {
		return nil
	}
	return strings.Split(strings.TrimRight(string(data), "\n"), "\n")
}

// Session returns the resource session metadata the host attaches to a plugin call.
func ResourceSession() resourcev2.Session {
	return resourcev2.Session{RequestID: uuid.NewString(), RequesterID: "zz-proof"}
}

// Ctx returns ctx carrying a resource session, as a host call does.
func Ctx(ctx context.Context) context.Context {
	return resourcev2.WithSession(ctx, ResourceSession())
}

// Connect loads the connections the plugin finds in $HOME/.kube/config and returns the one named name.
func (p *Plugin) Connect(ctx context.Context, name string) (types.Connection, []types.Connection, error) {
	conns, err := p.Resource.LoadConnections(Ctx(ctx))
	if err != nil {
		return types.Connection{}, nil, fmt.Errorf("LoadConnections: %w", err)
	}
	for _, c := range conns {
		if c.Name == name {
			return c, conns, nil
		}
	}
	names := make([]string, 0, len(conns))
	for _, c := range conns {
		names = append(names, c.Name)
	}
	return types.Connection{}, conns, fmt.Errorf("connection %q not found among %v", name, names)
}

func copyFile(src, dst string, mode os.FileMode) error {
	data, err := os.ReadFile(src)
	if err != nil {
		return err
	}
	return os.WriteFile(dst, data, mode)
}

// stampWriter writes every complete line it is given with a RFC3339Nano prefix of the harness clock.
type stampWriter struct {
	mu     sync.Mutex
	w      io.Writer
	closer io.Closer
	buf    bytes.Buffer
}

func (s *stampWriter) Write(p []byte) (int, error) {
	s.mu.Lock()
	defer s.mu.Unlock()
	s.buf.Write(p)
	for {
		i := bytes.IndexByte(s.buf.Bytes(), '\n')
		if i < 0 {
			break
		}
		line := s.buf.Next(i + 1)
		if _, err := fmt.Fprintf(s.w, "%s | %s", time.Now().UTC().Format("2006-01-02T15:04:05.000000Z"), line); err != nil {
			return len(p), err
		}
	}
	return len(p), nil
}

func (s *stampWriter) Close() {
	s.mu.Lock()
	defer s.mu.Unlock()
	if s.buf.Len() > 0 {
		_, _ = fmt.Fprintf(s.w, "%s | %s\n", time.Now().UTC().Format("2006-01-02T15:04:05.000000Z"), s.buf.String())
		s.buf.Reset()
	}
	if s.closer != nil {
		_ = s.closer.Close()
		s.closer = nil
	}
}
